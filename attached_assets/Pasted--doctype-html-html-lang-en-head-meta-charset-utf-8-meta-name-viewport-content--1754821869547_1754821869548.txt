    const $ = (sel) => document.querySelector(sel);
    const $$ = (sel) => Array.from(document.querySelectorAll(sel));

    function setLang(lang, { rerender = true } = {}) {
      const dict = I18N[lang] || I18N.en;
      $$('[data-i18n]').forEach(el => {
        const key = el.getAttribute('data-i18n');
        if (dict[key]) el.textContent = dict[key];
      });
      document.documentElement.lang = lang;
      const langToggle = $('#langToggle');
      if (langToggle) langToggle.textContent = lang === 'en' ? 'FR' : 'EN';
      localStorage.setItem('lang', lang);
      if (rerender) {
        renderWines();
        renderJournal();
      }
    }

    function wineCard(w, lang) {
      const t = w[lang] || w.en;
      return `
        <article class="reveal group rounded-3xl border border-stone-200 bg-white p-6 shadow-soft">
          <a href="${w.tech}" class="block" aria-label="View wine: ${t.name}">
            <div class="aspect-[3/4] overflow-hidden rounded-2xl bg-paper">
              <img src="${w.img}" alt="${t.name} bottle" class="h-full w-full object-contain group-hover:scale-[1.02] transition" loading="lazy" decoding="async" />
            </div>
            <div class="mt-4 flex items-center justify-between">
              <h3 class="font-serif text-xl">${t.name}</h3>
              <span class="text-sm text-ink/60">${t.vintage}</span>
            </div>
            <p class="mt-2 text-sm text-ink/70">${t.note}</p>
          </a>
        </article>`;
    }

    function renderWines() {
      const grid = $('#winesGrid');
      if (!grid) return;
      const lang = document.documentElement.lang || 'en';
      grid.innerHTML = DATA.wines.map(w => wineCard(w, lang)).join('');
      // reapply reveal on newly inserted cards
      $$('.reveal').forEach(el => io ? io.observe(el) : el.classList.add('visible'));
    }

    function journalCard(post, lang) {
      const t = post[lang] || post.en;
      const date = new Date(post.date);
      const label = date.toLocaleDateString(undefined, { month: 'short', year: 'numeric' });
      return `
        <article class="reveal rounded-3xl border border-stone-200 bg-white p-6 shadow-soft">
          <a href="${post.href}" class="block" aria-label="Read: ${t.title}">
            <div class="aspect-[16/10] overflow-hidden rounded-2xl bg-paper">
              <img src="${post.img}" alt="${t.title}" class="h-full w-full object-cover" loading="lazy" decoding="async" />
            </div>
            <div class="mt-4 flex items-center justify-between">
              <h3 class="font-serif text-xl">${t.title}</h3>
              <span class="text-sm text-ink/60">${label}</span>
            </div>
            <p class="mt-2 text-sm text-ink/70">${t.excerpt}</p>
          </a>
        </article>`;
    }
